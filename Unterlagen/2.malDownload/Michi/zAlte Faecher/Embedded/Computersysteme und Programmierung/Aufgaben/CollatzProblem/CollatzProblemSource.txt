#include <stdio.h>

void converge(int, int*, int*);

int main() {
	//N größte Startzahl, stepMin Minimale Schrittanzahl, max größte erreichte Zahl
	int n = 0, stepMin = 0, max = 0;

	do {
		fflush(stdin);
		printf("Please enter highest start number and minimal step count:\n");
	}while(scanf_s("%d %d", &n, &stepMin) != 2);

	int i;
	for (i = 1; i <= n; i++) {
		converge(i, &stepMin, &max);
	};
	printf("Biggest number was: %d", max);

	return 0;
}



void converge(int n, int* step, int* max) {
	int stepCounter = 0;
	int k = n;

	do{
		if (k > *max) {
			*max = k;
		}
		if(k == 1){
		//nothing
		}
		else if (k % 2 == 0) {
			k /= 2;
		}
		else {
			k = k * 3 + 1;
		}
		stepCounter++;
	}while(k > 1);

	if (stepCounter >= *step)
		printf("Number: %d was reached!\n", n);
};
