// Pin schreiben

#define GPIO_OUT_W1TS_REG__ 		(0x0008)
#define LED 						6
#define GPIO_OUT_W1TC_REG__			0x000C

volatile uint32_t* gpOutSetRegister = (volatile uint32_t*) ((GPIO_BASE_ADDRESS__) | (GPIO_OUT_W1TS_REG__));
volatile uint32_t* gpOutClearRegister = (volatile uint32_t*) ((GPIO_BASE_ADDRESS__) | (GPIO_OUT_W1TC_REG__));


// im programmteil
(*gpOutSetRegister) = (1 << LED);	//einschalten
(*gpOutClearRegister) = (1 << LED);	//ausschalten


//als funktion
void setOutput(uint32_t pinNumber){
	(*gpOutSetRegister) = (1 << pinNumber);
}

void clearOutput(uint32_t pinNumber){
	(*gpOutClearRegister) = (1 << pinNumber);	
}

while(1){
	setOutput(LED);
	vTaskDelay(100);
	clearOutput(LED);
	vTaskDelay(100);
}
