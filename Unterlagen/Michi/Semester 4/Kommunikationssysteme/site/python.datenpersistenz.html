    class PersistentObject:

    """Objekt, das in ein Text-File geschrieben werden kann (zum Beispiel bei Programmende ) und bei Programmstart aus diesem File wieder erzeugt werden kann.

    Es wird nicht das ganze Objekt geschrieben, sondern *das* Dict, das die Attribute hält.

    Beim Wiederherstellen, wird das Dict aus dem File gelesen. Jetzt ist es aber nur Text.
    Ein Dict wird daraus erst durch die Anwendung von ``eval()``. Wenn es dann ein "echtes"
    Dict ist, wird *das* Dict des Objekts mit diesem von :term:`HDD` gelesenen Dict
    aktualisiert.

    PARAMETERS:
        id (str):
            Identifikation (Name) des Objekts.
    """

    def __init__( self, id: str):
        self.__id = id
        self.__data = 42
        return

    def data( self, arg=None):
        """Lesen/Schreiben der Daten des Objekts.
        """
        if arg is None:
            return self.__data

        self.__data = arg
        return self

    def id( self) -> str:
        """Identifikation (Name) des Objekts.
        """
        return self.__id

    def pathname( self) -> str:
        """Name des Pfades des Files, in den das Objekt geschrieben bzw von dem es gelesen werden soll.
        """
        return "./" + self.id() + ".txt"

    def restore( self) -> bool:
        """Wiederherstellen (Lesen) des Objekts aus den Daten des Files, dessen Name :meth:`pathname` liefert.

        **Genau genommen** wird nicht das Objekt wiederhergestellt, sondern nur seine Attribute.
        Dafür muss das Objekt aber bereits existieren. Um wirklich ein Objekt wiederherzustellen,
        ohne es zuerst anlegen zu müssen, um dann "nur" seine Attribute wiederherzustellen, geht
        man vor wie gezeigt in :class:`PersistentObject2`.
        """
        with open( self.pathname(), "r") as f:
            try:
                d = eval( f.read())
                self.__dict__.update( d)
                return True

            except SyntaxError:
                pass

            return False

    def store( self) -> bool:
        """Schreiben des Objekts in das File, dessen Name :meth:`pathname` liefert.

        **Genau genommen** wird nicht das Objekt geschrieben, sondern nur seine Attribute, s. :meth:`restore`.
        """
        with open( self.pathname(), "w") as f:
            f.write( str( self.__dict__))

        return True

    class PersistentObject2( PersistentObject):

    """Wie :class:`PersistentObject`, nur dass hier das gesamte Objekt persistiert und dann auch wiederhergestellt wird.
    """

    @staticmethod
    def PathnameFromId( id: str) -> str:
        """
        """
        return "./" + id + ".txt"

    @staticmethod
    def Restore( id: str):
        """
        """
        this = pickle.load( open( PersistentObject2.PathnameFromId( id), "rb"))
        return this

    @staticmethod
    def Store( this) -> str:
        """
        """
        pickle.dump( this, open( this.pathname(), "wb"))
        return this.id()


    def __eq__( self, other) -> bool:
        if not self.id() == other.id():
            return False

        return self.data() == other.data()

    ...das Modul shelve 1 der Standard-Library! Was könnten Sie damit anfangen?

    Folgendes Beispiel mag etwas Licht in die Situation bringen:

    """Persistente Dictionaries.

Als Vorbereitung auf diese Übung lesen Sie bitte
https://docs.python.org/3/library/shelve.html aufmerksam durch.

Wir legen das Shelf in dieser Übung so an, dass die Inhalte auf Disk direkt mit denen
im Programm verbunden sind. Na ja, ein ``synch`` weit entfernt sind sie dann doch.
"""

import os, shelve

from pie import pieTime as pieT


if __name__ == "__main__":
    def _Lab_():
        N = 100_000
        pathname_of_shelf = "./the_shelffile.dat"

        ### Großes Shelf anlegen
        #
        if os.path.exists( pathname_of_shelf):
            os.unlink( pathname_of_shelf)

        with pieT.PieRuntimemeter( "Shelf mit %d Einträgen erzeugen" % N) as rtm:
            shelf = shelve.open( pathname_of_shelf, writeback=True)
            for i in range( N):
                shelf[ str( i)] = i

            shelf.close()

        rtm.report( print)

        ### Shelf erneut öffnen und schauen, ob noch alles da ist.
        #
        with pieT.PieRuntimemeter( "Shelf mit %d Einträgen öffnen und EINEN Eintrag lesen" % N) as rtm:
            shelf = shelve.open( "./the_shelffile.dat", writeback=True)
            data = shelf[ "42"]
            shelf.close()

        print( data)
        rtm.report( print)

        ### Shelf erneut öffnen und ändern
        #
        with pieT.PieRuntimemeter( "Shelf mit %d Einträgen öffnen und EINEN Eintrag ändern" % N) as rtm:
            shelf = shelve.open( "./the_shelffile.dat", writeback=True)
            shelf[ "42"] = "zweiundvierzig"
            shelf.close()

        print( data)
        rtm.report( print)

        with pieT.PieRuntimemeter( "Shelf mit %d Einträgen öffnen und EINEN Eintrag lesen" % N) as rtm:
            shelf = shelve.open( "./the_shelffile.dat", writeback=True)
            data = shelf[ "42"]
            assert data == "zweiundvierzig"
            shelf.close()

        print( data)
        rtm.report( print)
        return

    _Lab_()
